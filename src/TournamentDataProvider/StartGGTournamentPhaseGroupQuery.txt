query TournamentPhaseGroupQuery($id: ID!) {
	phaseGroup(id: $id) {
		seedMap
		seeds(query: {page: 1, perPage: 500}) {
			nodes {
				seedNum
				progressionSource {
					originPhaseGroup {
						id
					}
				}
				entrant {
					id
					name
					participants {
						player {
							id
							gamerTag
							prefix
							name
						}
						user {
							id
							prefix
							name
							location {
								country
								state
								city
							}
						}
					}
				}
			}
		}
		sets(page: 1, perPage: 9999, filters: {showByes: true, hideEmpty: false}, sortType: CALL_ORDER){
			nodes {
				id
				round
				identifier
				entrant1Score
				entrant2Score
				state
				slots {
					prereqType
					prereqId
					prereqPlacement
					seed {
						seedNum
					}
				}
			}
		}
		progressionsOut {
			id
		}
	}
}
